streamlit
plotly
pandas


st.set_page_config(
    page_title="SmartSpend AI",
    page_icon="💰",
    layout="wide"
)

# ---------- DATA ----------
if "expenses" not in st.session_state:
    st.session_state.expenses = pd.DataFrame(
        columns=["Date", "Description", "Amount", "Category"]
    )

# ---------- AI CATEGORY ----------
def categorize_expense(description):
    text = description.lower()

    if any(word in text for word in ["food", "pizza", "burger", "restaurant", "dosa", "coffee"]):
        return "Food"
    elif any(word in text for word in ["uber", "ola", "metro", "bus", "petrol", "fuel"]):
        return "Transport"
    elif any(word in text for word in ["amazon", "flipkart", "shopping", "clothes"]):
        return "Shopping"
    elif any(word in text for word in ["movie", "netflix", "game", "entertainment"]):
        return "Entertainment"
    elif any(word in text for word in ["book", "course", "college", "education"]):
        return "Education"
    elif any(word in text for word in ["rent", "electricity", "water", "recharge"]):
        return "Bills"
    else:
        return "Other"

# ---------- HEADER ----------
st.title("💰 SmartSpend AI")
st.subheader("AI Financial Assistant for Smart Expense Management")

st.write(
    "Track your expenses, automatically categorize transactions, "
    "and understand your spending patterns."
)

# ---------- ADD EXPENSE ----------
st.header("➕ Add Expense")

col1, col2, col3 = st.columns(3)

with col1:
    expense_date = st.date_input("Date", date.today())

with col2:
    description = st.text_input(
        "Expense Description",
        placeholder="Example: Pizza, Uber, Amazon"
    )

with col3:
    amount = st.number_input(
        "Amount (₹)",
        min_value=0.0,
        step=10.0
    )

if st.button("Add Expense", type="primary"):
    if description.strip() == "":
        st.warning("Please enter an expense description.")
    elif amount <= 0:
        st.warning("Please enter a valid amount.")
    else:
        category = categorize_expense(description)

        new_expense = pd.DataFrame({
            "Date": [expense_date],
            "Description": [description],
            "Amount": [amount],
            "Category": [category]
        })

        st.session_state.expenses = pd.concat(
            [st.session_state.expenses, new_expense],
            ignore_index=True
        )

        st.success(f"Expense added and categorized as {category}!")

# ---------- DASHBOARD ----------
st.header("📊 Financial Dashboard")

df = st.session_state.expenses

if len(df) > 0:

    total_spending = df["Amount"].sum()
    transaction_count = len(df)

    category_totals = (
        df.groupby("Category")["Amount"]
        .sum()
        .sort_values(ascending=False)
    )

    top_category = category_totals.index[0]

    c1, c2, c3 = st.columns(3)

    c1.metric("Total Spending", f"₹{total_spending:,.2f}")
    c2.metric("Transactions", transaction_count)
    c3.metric("Top Spending Category", top_category)

    st.subheader("📋 Expense History")
    st.dataframe(df, use_container_width=True)

    st.subheader("📈 Spending by Category")

    chart_data = category_totals.reset_index()
    chart_data.columns = ["Category", "Amount"]

    st.bar_chart(
        chart_data.set_index("Category")
    )

    # ---------- AI INSIGHTS ----------
    st.header("🤖 AI Spending Insights")

    average_expense = df["Amount"].mean()

    st.write(
        f"• Your average transaction is **₹{average_expense:,.2f}**."
    )

    st.write(
        f"• Your highest spending category is **{top_category}**."
    )

    if total_spending > 10000:
        st.warning(
            "⚠️ Your total spending has crossed ₹10,000. "
            "Consider reviewing your high-value expenses."
        )
    else:
        st.success(
            "✅ Your current spending is below ₹10,000."
        )

    if category_totals.iloc[0] > total_spending * 0.5:
        st.info(
            f"💡 More than 50% of your recorded spending is in "
            f"**{top_category}**. Consider monitoring this category."
        )

else:

    st.info(
        "No expenses added yet. Add your first expense above "
        "to see your financial dashboard."
    )
